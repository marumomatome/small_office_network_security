! Small Office Network - Router Configuration
! Device: Office-Router
! Platform: Cisco 2911

enable
configure terminal

hostname Office-Router

interface gigabitEthernet 0/0
 ip address 192.168.10.1 255.255.255.0
 no shutdown
exit

enable secret **********
service password-encryption

ip domain-name myoffice.local
username netadmin secret *******
crypto key generate rsa
ip ssh version 2

line vty 0 4
 login local
 transport input ssh
exit

end

copy running-config startup-config

show ip interface brief
show running-config
